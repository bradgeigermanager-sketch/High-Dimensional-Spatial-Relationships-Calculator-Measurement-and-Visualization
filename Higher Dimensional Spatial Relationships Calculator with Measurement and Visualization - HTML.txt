        // ================= 3D GRAPH CANVAS INTERACTIVITY =================
        let graphCanvas, graphCtx;


        function initGraphCanvas() {
            graphCanvas = document.getElementById('graphCanvas');
            if (!graphCanvas) return;
            graphCtx = graphCanvas.getContext('2d');


            // 3D Orbiting Drag Events
            graphCanvas.addEventListener('mousedown', (e) => {
                state.graph.isDragging = true;
                state.graph.lastMouseX = e.clientX;
                state.graph.lastMouseY = e.clientY;
            });


            window.addEventListener('mousemove', (e) => {
                if (!state.graph.isDragging) return;
                const dx = e.clientX - state.graph.lastMouseX;
                const dy = e.clientY - state.graph.lastMouseY;
                state.graph.rotY += dx * 0.008;
                state.graph.rotX += dy * 0.008;
                // Clamp elevation to avoid gimbal flip
                state.graph.rotX = Math.max(-Math.PI / 2 + 0.1, Math.min(Math.PI / 2 - 0.1, state.graph.rotX));
                state.graph.lastMouseX = e.clientX;
                state.graph.lastMouseY = e.clientY;
                renderGraph();
            });


            window.addEventListener('mouseup', () => {
                state.graph.isDragging = false;
            });


            // Touch support for 3D Orbit
            graphCanvas.addEventListener('touchstart', (e) => {
                if (e.touches.length > 0) {
                    state.graph.isDragging = true;
                    state.graph.lastMouseX = e.touches[0].clientX;
                    state.graph.lastMouseY = e.touches[0].clientY;
                }
            }, { passive: true });


            graphCanvas.addEventListener('touchmove', (e) => {
                if (!state.graph.isDragging || e.touches.length === 0) return;
                const dx = e.touches[0].clientX - state.graph.lastMouseX;
                const dy = e.touches[0].clientY - state.graph.lastMouseY;
                state.graph.rotY += dx * 0.008;
                state.graph.rotX += dy * 0.008;
                state.graph.lastMouseX = e.touches[0].clientX;
                state.graph.lastMouseY = e.touches[0].clientY;
                renderGraph();
            }, { passive: true });


            window.addEventListener('touchend', () => { state.graph.isDragging = false; });


            resizeGraphCanvas();
        }

        function syncGraphFromCalculator() {
            try {
                const inputA = document.getElementById('vectorAInput');
                const inputB = document.getElementById('vectorBInput');
                if (!inputA || !inputB) return;


                state.vectorA = parseVector(inputA.value);
                state.vectorB = parseVector(inputB.value);

                const badge = document.getElementById('highDimBadge');
                if (badge) badge.textContent = `Dim: ${N}D`;


                // Populate 3D Dimension Selectors
                const selectX = document.getElementById('selectXDim');
                const selectY = document.getElementById('selectYDim');
                const selectZ = document.getElementById('selectZDim');


                if (selectX && selectY && selectZ) {
                    selectX.innerHTML = '';
                    selectY.innerHTML = '';
                    selectZ.innerHTML = '';


                    for (let i = 0; i < N; i++) {
                        const label = `Dimension ${i + 1} (u: ${state.vectorA[i].toFixed(2)}, v: ${state.vectorB[i].toFixed(2)})`;
                        
                        const optX = document.createElement('option');
                        optX.value = i; optX.textContent = label;
                        if (i === 0) optX.selected = true;
                        selectX.appendChild(optX);


                        const optY = document.createElement('option');
                        optY.value = i; optY.textContent = label;
                        if (i === (N > 1 ? 1 : 0)) optY.selected = true;
                        selectY.appendChild(optY);


                        const optZ = document.createElement('option');
                        optZ.value = i; optZ.textContent = label;
                        if (i === (N > 2 ? 2 : (N > 1 ? 1 : 0))) optZ.selected = true;
                        selectZ.appendChild(optZ);
                    }
                }


                state.graph.axisX = 0;
                state.graph.axisY = N > 1 ? 1 : 0;
                state.graph.axisZ = N > 2 ? 2 : (N > 1 ? 1 : 0);


                updateDimensionMapping();
            } catch (err) {
                console.warn("Sync graph error:", err);
            }
        }

            updateInputFieldsFrom3D();
            renderGraph();
        }

        // Projects high-dimensional vectors to 3D basis coordinates
        function projectVectorTo3D(vec) {
            const N = vec.length;
            if (N === 0) return { x: 0, y: 0, z: 0 };


            let x = vec[state.graph.axisX % N] || 0;
            let y = vec[state.graph.axisY % N] || 0;
            let z = vec[state.graph.axisZ % N] || 0;


            if (state.graph.collapseMode === 'l2norm') {
                let sumSq = 0;
                for (let i = 0; i < N; i++) {
                    if (i !== state.graph.axisX % N && i !== state.graph.axisY % N) {
                        sumSq += vec[i] * vec[i];
                    }
                }
                z = Math.sqrt(sumSq);
            } else if (state.graph.collapseMode === 'mean') {
                let sum = 0, count = 0;
                for (let i = 0; i < N; i++) {
                    if (i !== state.graph.axisX % N && i !== state.graph.axisY % N) {
                        sum += vec[i];
                        count++;
                    }
                }
                z = count > 0 ? sum / count : 0;
            } else if (state.graph.collapseMode === 'sum') {
                let sum = 0;
                for (let i = 0; i < N; i++) {
                    if (i !== state.graph.axisX % N && i !== state.graph.axisY % N) {
                        sum += vec[i];
                    }
                }
                z = sum;
            }


            if (state.graph.radialMode === 'dim4' && N >= 4) {
                const r = Math.abs(vec[3]);
                const currentRadius = Math.hypot(x, y, z) || 1;
                x = (x / currentRadius) * r;
                y = (y / currentRadius) * r;
                z = (z / currentRadius) * r;
            } else if (state.graph.radialMode === 'fullNorm') {
                let normSq = 0;
                for (let i = 0; i < N; i++) normSq += vec[i] * vec[i];
                const r = Math.sqrt(normSq);
                const currentRadius = Math.hypot(x, y, z) || 1;
                x = (x / currentRadius) * r;
                y = (y / currentRadius) * r;
                z = (z / currentRadius) * r;
            }


            return { x, y, z };
        }


        // 3D Point Projection to 2D Canvas Viewport (Standard Right-Handed Camera Matrix)
        function project3Dto2D(point3d, center, scale) {
            const cosY = Math.cos(state.graph.rotY);
            const sinY = Math.sin(state.graph.rotY);
            const cosX = Math.cos(state.graph.rotX);
            const sinX = Math.sin(state.graph.rotX);


            // Yaw (Rotation around Y-axis)
            const x1 = point3d.x * cosY + point3d.z * sinY;
            const y1 = point3d.y;
            const z1 = -point3d.x * sinY + point3d.z * cosY;


            // Pitch (Rotation around X-axis)
            const x2 = x1;
            const y2 = y1 * cosX - z1 * sinX;
            const z2 = y1 * sinX + z1 * cosX;


            return {
                x: center.x + x2 * scale,
                y: center.y - y2 * scale, // Invert Y for canvas coordinate system
                z: z2 // Depth for sorting
            };
        }


        function renderGraph() {
            if (!graphCtx || !graphCanvas) return;
            const w = graphCanvas.width;
            const h = graphCanvas.height;
            const center = { x: w / 2, y: h / 2 };
            const scale = (Math.min(w, h) / 8) * state.graph.zoom;
            const isDark = document.documentElement.classList.contains('dark');


            graphCtx.clearRect(0, 0, w, h);


            // Origin
            const origin2D = project3Dto2D({ x: 0, y: 0, z: 0 }, center, scale);


            // Draw 3D Axes (X: Red, Y: Green, Z: Blue/Purple)
            const axisLength = 3;
            const axisX2D = project3Dto2D({ x: axisLength, y: 0, z: 0 }, center, scale);
            const axisY2D = project3Dto2D({ x: 0, y: axisLength, z: 0 }, center, scale);
            const axisZ2D = project3Dto2D({ x: 0, y: 0, z: axisLength }, center, scale);


            // Floor Grid Plane (XZ plane at y=0)
            graphCtx.strokeStyle = isDark ? 'rgba(255, 255, 255, 0.08)' : 'rgba(0, 0, 0, 0.08)';
            graphCtx.lineWidth = 1 * window.devicePixelRatio;
            for (let i = -3; i <= 3; i++) {
                const p1 = project3Dto2D({ x: i, y: 0, z: -3 }, center, scale);
                const p2 = project3Dto2D({ x: i, y: 0, z: 3 }, center, scale);
                graphCtx.beginPath(); graphCtx.moveTo(p1.x, p1.y); graphCtx.lineTo(p2.x, p2.y); graphCtx.stroke();


                const p3 = project3Dto2D({ x: -3, y: 0, z: i }, center, scale);
                const p4 = project3Dto2D({ x: 3, y: 0, z: i }, center, scale);
                graphCtx.beginPath(); graphCtx.moveTo(p3.x, p3.y); graphCtx.lineTo(p4.x, p4.y); graphCtx.stroke();
            }


            // Draw Axes Lines
            draw3DLine(graphCtx, origin2D, axisX2D, '#ef4444', 2, 'X');
            draw3DLine(graphCtx, origin2D, axisY2D, '#10b981', 2, 'Y');
            draw3DLine(graphCtx, origin2D, axisZ2D, '#a855f7', 2, 'Z');


            const u3d = state.graph.u3d;
            const v3d = state.graph.v3d;


            const u2d = project3Dto2D(u3d, center, scale);
            const v2d = project3Dto2D(v3d, center, scale);


            // Bounding Wireframe Box
            const chkBox = document.getElementById('chkBoundingBox');
            if (chkBox && chkBox.checked) {
                drawWireBox(graphCtx, u3d, center, scale, 'rgba(59, 130, 246, 0.15)');
                drawWireBox(graphCtx, v3d, center, scale, 'rgba(16, 185, 129, 0.15)');
            }


            // Drop lines to XZ floor plane (y=0)
            const chkProj = document.getElementById('chkProjection');
            if (chkProj && chkProj.checked) {
                const uFloor = project3Dto2D({ x: u3d.x, y: 0, z: u3d.z }, center, scale);
                const vFloor = project3Dto2D({ x: v3d.x, y: 0, z: v3d.z }, center, scale);


                drawDashedLine(graphCtx, u2d, uFloor, 'rgba(59, 130, 246, 0.5)');
                drawDashedLine(graphCtx, v2d, vFloor, 'rgba(16, 185, 129, 0.5)');
            }


            // 3D Taxicab Path
            const chkMan = document.getElementById('chkManhattan');
            if (chkMan && chkMan.checked) {
                const pStep1 = project3Dto2D({ x: v3d.x, y: u3d.y, z: u3d.z }, center, scale);
                const pStep2 = project3Dto2D({ x: v3d.x, y: v3d.y, z: u3d.z }, center, scale);


                drawDashedLine(graphCtx, u2d, pStep1, '#a855f7');
                drawDashedLine(graphCtx, pStep1, pStep2, '#a855f7');
                drawDashedLine(graphCtx, pStep2, v2d, '#a855f7');
            }


            // Vector Projection onto Vector A
            const chkVecProj = document.getElementById('chkVectorProj');
            const normU2 = u3d.x * u3d.x + u3d.y * u3d.y + u3d.z * u3d.z;
            if (chkVecProj && chkVecProj.checked && normU2 > 1e-6) {
                const dot = u3d.x * v3d.x + u3d.y * v3d.y + u3d.z * v3d.z;
                const scalarProj = dot / normU2;
                const projPoint3d = {
                    x: u3d.x * scalarProj,
                    y: u3d.y * scalarProj,
                    z: u3d.z * scalarProj
                };
                const proj2d = project3Dto2D(projPoint3d, center, scale);
                drawDashedLine(graphCtx, v2d, proj2d, '#10b981');
                draw3DVectorNode(graphCtx, origin2D, proj2d, '#34d399', '');
            }


            // 3D Angle Arc
            const chkAngle = document.getElementById('chkAngleArc');
            const normU = Math.sqrt(normU2);
            const normV = Math.sqrt(v3d.x * v3d.x + v3d.y * v3d.y + v3d.z * v3d.z);
            if (chkAngle && chkAngle.checked && normU > 1e-6 && normV > 1e-6) {
                const steps = 20;
                graphCtx.beginPath();
                for (let i = 0; i <= steps; i++) {
                    const t = i / steps;
                    const interpPoint = {
                        x: (1 - t) * u3d.x / normU + t * v3d.x / normV,
                        y: (1 - t) * u3d.y / normU + t * v3d.y / normV,
                        z: (1 - t) * u3d.z / normU + t * v3d.z / normV
                    };
                    const normI = Math.sqrt(interpPoint.x ** 2 + interpPoint.y ** 2 + interpPoint.z ** 2) || 1;
                    const arc3d = {
                        x: (interpPoint.x / normI) * 0.8,
                        y: (interpPoint.y / normI) * 0.8,
                        z: (interpPoint.z / normI) * 0.8
                    };
                    const arc2d = project3Dto2D(arc3d, center, scale);
                    if (i === 0) graphCtx.moveTo(arc2d.x, arc2d.y);
                    else graphCtx.lineTo(arc2d.x, arc2d.y);
                }
                graphCtx.strokeStyle = '#f59e0b';
                graphCtx.lineWidth = 2 * window.devicePixelRatio;
                graphCtx.stroke();
            }


            // 3D Euclidean Distance Line
            const chkEuc = document.getElementById('chkEuclidean');
            if (chkEuc && chkEuc.checked) {
                graphCtx.beginPath();
                graphCtx.moveTo(u2d.x, u2d.y);
                graphCtx.lineTo(v2d.x, v2d.y);
                graphCtx.strokeStyle = '#6366f1';
                graphCtx.lineWidth = 2.5 * window.devicePixelRatio;
                graphCtx.stroke();
            }


            // Vector A Arrow in 3D
            draw3DVectorNode(graphCtx, origin2D, u2d, '#3b82f6', 'Vector A');
            // Vector B Arrow in 3D
            draw3DVectorNode(graphCtx, origin2D, v2d, '#10b981', 'Vector B');


            // Calculate 3D Projected Distance Values
            const dx = u3d.x - v3d.x;
            const dy = u3d.y - v3d.y;
            const dz = u3d.z - v3d.z;


            const euclid3D = Math.sqrt(dx * dx + dy * dy + dz * dz);
            const manhattan3D = Math.abs(dx) + Math.abs(dy) + Math.abs(dz);


            const dot3D = u3d.x * v3d.x + u3d.y * v3d.y + u3d.z * v3d.z;
            const cos3D = (normU && normV) ? Math.max(-1, Math.min(1, dot3D / (normU * normV))) : 0;
            const angle3D = Math.acos(cos3D) * (180 / Math.PI);


            const coordA = document.getElementById('graphCoordA');
            const coordB = document.getElementById('graphCoordB');
            const valEuc = document.getElementById('graphValEuclid');
            const valMan = document.getElementById('graphValManhattan');
            const valAng = document.getElementById('graphValAngle');


            if (coordA) coordA.textContent = `[${u3d.x.toFixed(2)}, ${u3d.y.toFixed(2)}, ${u3d.z.toFixed(2)}]`;
            if (coordB) coordB.textContent = `[${v3d.x.toFixed(2)}, ${v3d.y.toFixed(2)}, ${v3d.z.toFixed(2)}]`;
            if (valEuc) valEuc.textContent = euclid3D.toFixed(3);
            if (valMan) valMan.textContent = manhattan3D.toFixed(3);
            if (valAng) valAng.textContent = `${angle3D.toFixed(1)}°`;
        }


        function draw3DLine(ctx, p1, p2, color, width, label) {
            ctx.beginPath();
            ctx.moveTo(p1.x, p1.y);
            ctx.lineTo(p2.x, p2.y);
            ctx.strokeStyle = color;
            ctx.lineWidth = width * window.devicePixelRatio;
            ctx.stroke();


            if (label) {
                ctx.fillStyle = color;
                ctx.font = `${10 * window.devicePixelRatio}px JetBrains Mono`;
                ctx.fillText(label, p2.x + 5, p2.y + 5);
            }
        }


        function drawDashedLine(ctx, p1, p2, color) {
            ctx.beginPath();
            ctx.moveTo(p1.x, p1.y);
            ctx.lineTo(p2.x, p2.y);
            ctx.strokeStyle = color;
            ctx.lineWidth = 1.5 * window.devicePixelRatio;
            ctx.setLineDash([4, 4]);
            ctx.stroke();
            ctx.setLineDash([]);
        }


        function drawWireBox(ctx, point3d, center, scale, color) {
            const corners = [
                { x: 0, y: 0, z: 0 }, { x: point3d.x, y: 0, z: 0 },
                { x: point3d.x, y: point3d.y, z: 0 }, { x: 0, y: point3d.y, z: 0 },
                { x: 0, y: 0, z: point3d.z }, { x: point3d.x, y: 0, z: point3d.z },
                { x: point3d.x, y: point3d.y, z: point3d.z }, { x: 0, y: point3d.y, z: point3d.z }
            ].map(p => project3Dto2D(p, center, scale));


            ctx.strokeStyle = color;
            ctx.lineWidth = 1 * window.devicePixelRatio;


            const edges = [
                [0, 1], [1, 2], [2, 3], [3, 0],
                [4, 5], [5, 6], [6, 7], [7, 4],
                [0, 4], [1, 5], [2, 6], [3, 7]
            ];


            edges.forEach(([i, j]) => {
                ctx.beginPath();
                ctx.moveTo(corners[i].x, corners[i].y);
                ctx.lineTo(corners[j].x, corners[j].y);
                ctx.stroke();
            });
        }


        function draw3DVectorNode(ctx, origin, target, color, label) {
            ctx.beginPath();
            ctx.moveTo(origin.x, origin.y);
            ctx.lineTo(target.x, target.y);
            ctx.strokeStyle = color;
            ctx.lineWidth = 3 * window.devicePixelRatio;
            ctx.stroke();


            // Head Sphere Node
            ctx.beginPath();
            ctx.arc(target.x, target.y, 7 * window.devicePixelRatio, 0, Math.PI * 2);
            ctx.fillStyle = color;

        // ================= VECTOR CALCULATOR COMPUTATIONS =================
        function parseVector(text) {
            const parts = text.trim().split(/[\s,]+/).filter(x => x !== "");
            const arr = new Float64Array(parts.length);
            for (let i = 0; i < parts.length; i++) {
                const val = parseFloat(parts[i]);
                if (isNaN(val)) throw new Error(`Invalid number at position ${i + 1}`);
                arr[i] = val;
            }
            return arr;
        }


        function runCalculation() {
            const errBanner = document.getElementById('calcError');
            if (errBanner) errBanner.classList.add('hidden');


            try {
                const inputA = document.getElementById('vectorAInput');
                const inputB = document.getElementById('vectorBInput');
                if (!inputA || !inputB) return;


                state.vectorA = parseVector(inputA.value);
                state.vectorB = parseVector(inputB.value);


                const badgeA = document.getElementById('dimBadgeA');
                const badgeB = document.getElementById('dimBadgeB');
                if (badgeA) badgeA.textContent = `Dim: ${state.vectorA.length}`;
                if (badgeB) badgeB.textContent = `Dim: ${state.vectorB.length}`;


                if (state.vectorA.length !== state.vectorB.length) {
                    throw new Error(`Dimension Mismatch: Vector A ({state.vectorB.length}D)`);
                }


                const N = state.vectorA.length;
                if (N === 0) throw new Error("Vectors cannot be empty.");


                // Single-pass compute loop
                let dot = 0, normA2 = 0, normB2 = 0, manhattan = 0, euclidean2 = 0;
                for (let i = 0; i < N; i++) {
                    const a = state.vectorA[i];
                    const b = state.vectorB[i];
                    const diff = a - b;


                    dot += a * b;
                    normA2 += a * a;
                    normB2 += b * b;
                    manhattan += Math.abs(diff);
                    euclidean2 += diff * diff;
                }


                const normA = Math.sqrt(normA2);
                const normB = Math.sqrt(normB2);
                const euclidean = Math.sqrt(euclidean2);


                const EPS = 1e-12;
                let cosSim = 0;
                if (normA > EPS && normB > EPS) {
                    cosSim = Math.max(-1, Math.min(1, dot / (normA * normB)));
                }


                const cosDist = 1 - cosSim;
                const angleDeg = Math.acos(cosSim) * (180 / Math.PI);


                state.calcResult = { N, dot, normA, normB, euclidean, manhattan, cosSim, cosDist, angleDeg };


                // Update UI Display
                const normElA = document.getElementById('normValA');
                const normElB = document.getElementById('normValB');
                if (normElA) normElA.textContent = normA.toFixed(4);
                if (normElB) normElB.textContent = normB.toFixed(4);


                const resSim = document.getElementById('resCosineSim');
                const resEuc = document.getElementById('resEuclidean');
                const resCosD = document.getElementById('resCosDist');
                const resMan = document.getElementById('resManhattan');
                const resDotEl = document.getElementById('resDot');
                const resAngEl = document.getElementById('resAngle');


                if (resSim) resSim.textContent = cosSim.toFixed(4);
                if (resEuc) resEuc.textContent = euclidean.toFixed(4);
                if (resCosD) resCosD.textContent = cosDist.toFixed(4);
                if (resMan) resMan.textContent = manhattan.toFixed(2);
                if (resDotEl) resDotEl.textContent = dot.toFixed(4);
                if (resAngEl) resAngEl.textContent = `${angleDeg.toFixed(1)}°`;


                // Update Visual Progress Bars
                const barCos = document.getElementById('barCosine');
                const barEuc = document.getElementById('barEuclidean');
                const barCosD = document.getElementById('barCosDist');


                if (barCos) barCos.style.width = `${Math.max(0, (cosSim + 1) / 2 * 100)}%`;
                if (barEuc) barEuc.style.width = `${Math.min(100, (euclidean / (Math.sqrt(N) || 1)) * 50)}%`;
                if (barCosD) barCosD.style.width = `${Math.min(100, (cosDist / 2) * 100)}%`;


                // Step-by-Step Equations Breakdown
                const sDot = document.getElementById('stepDot');
                const sNormA = document.getElementById('stepNormA');
                const sNormB = document.getElementById('stepNormB');
                const sCos = document.getElementById('stepCosineEq');
                const sEuc = document.getElementById('stepEuclidEq');


                if (sDot) sDot.innerHTML = `1. Dot Product () = <span class="text-brand-400 font-bold">${dot.toFixed(6)}</span>`;
                if (sNormA) sNormA.innerHTML = `2. $\|\mathbf{u}\|^2_2 = \sum u_i^2$ = ${normA2.toFixed(6)}  $\|\mathbf{u}\|_2 = <span class="text-brand-400 font-bold">${normA.toFixed(6)}</span>`;
                if (sNormB) sNormB.innerHTML = `3. $\|\mathbf{v}\|^2_2 = \sum v_i^2$ = ${normB2.toFixed(6)}  $\|\mathbf{v}\|_2 = <span class="text-emerald-400 font-bold">${normB.toFixed(6)}</span>`;
                if (sCos) sCos.innerHTML = `4. Cosine Sim = $\\frac{${dot.toFixed(4)}}{${(normA * normB).toFixed(4)}}$ = <span class="text-yellow-400 font-bold">${cosSim.toFixed(6)}</span>`;
                if (sEuc) sEuc.innerHTML = `5. Euclidean = $\\sqrt{${euclidean2.toFixed(6)}}$ = <span class="text-indigo-400 font-bold">${euclidean.toFixed(6)}</span>`;


                if (window.MathJax && window.MathJax.typesetPromise) {
                    window.MathJax.typesetPromise();
                }
            } catch (err) {
                if (errBanner) {
                    errBanner.classList.remove('hidden');
                    const msg = document.getElementById('calcErrorMsg');
                    if (msg) msg.textContent = err.message;
                }
            }
        }


        function generateRandomVectors(dim) {
            const vecA = new Float64Array(dim);
            const vecB = new Float64Array(dim);
            for (let i = 0; i < dim; i++) {
                vecA[i] = Number(((Math.random() - 0.5) * 2).toFixed(4));
                vecB[i] = Number(((Math.random() - 0.5) * 2).toFixed(4));
            }
            const inputA = document.getElementById('vectorAInput');
            const inputB = document.getElementById('vectorBInput');
            if (inputA) inputA.value = Array.from(vecA).join(', ');
            if (inputB) inputB.value = Array.from(vecB).join(', ');
            runCalculation();
        }


        function normalizeInputs() {
            try {
                runCalculation();
                const normA = state.calcResult.normA;
                const normB = state.calcResult.normB;
                const inputA = document.getElementById('vectorAInput');
                const inputB = document.getElementById('vectorBInput');


                if (normA > 0 && inputA) {
                    for (let i = 0; i < state.vectorA.length; i++) state.vectorA[i] /= normA;
                    inputA.value = Array.from(state.vectorA).map(x => x.toFixed(4)).join(', ');
                }
                if (normB > 0 && inputB) {
                    for (let i = 0; i < state.vectorB.length; i++) state.vectorB[i] /= normB;
                    inputB.value = Array.from(state.vectorB).map(x => x.toFixed(4)).join(', ');
                }
                runCalculation();
            } catch (err) {}
        }

        function setUmapPreset(preset) {
            state.umap.points = [];
            const count = 40;
            if (preset === 'hierarchical') {
                for (let i = 0; i < count; i++) {
                    const depth = (i % 4) + 1;
                    const angle = (Math.floor(i / 4) / (count / 4)) * Math.PI * 2;
                    const r = (depth / 4) * 0.75;
                    state.umap.points.push({
                        x: Math.cos(angle) * r,
                        y: Math.sin(angle) * r,
                        cluster: depth
                    });
                }
            } else {
                for (let c = 0; c < 3; c++) {
                    const cx = (c - 1) * 0.45;
                    const cy = (c % 2 === 0 ? 0.3 : -0.3);
                    for (let i = 0; i < 15; i++) {
                        state.umap.points.push({
                            x: cx + (Math.random() - 0.5) * 0.25,
                            y: cy + (Math.random() - 0.5) * 0.25,
                            cluster: c + 1
                        });
                    }
                }
            }
            updateUmapSimulation();
        }


        function updateUmapSimulation() {
            const inMetric = document.getElementById('umapInputMetric');
            const outMetric = document.getElementById('umapOutputMetric');
            const sliderN = document.getElementById('sliderNeighbors');
            const sliderK = document.getElementById('sliderCurvature');


            if (inMetric) state.umap.inputMetric = inMetric.value;
            if (outMetric) state.umap.outputMetric = outMetric.value;
            if (sliderN) state.umap.nNeighbors = parseInt(sliderN.value);
            if (sliderK) state.umap.curvature = parseFloat(sliderK.value);


            const valN = document.getElementById('valNeighbors');
            const valK = document.getElementById('valCurvature');
            if (valN) valN.textContent = state.umap.nNeighbors;
            if (valK) valK.textContent = state.umap.curvature.toFixed(1);


            // Generate Fuzzy Simplicial Edges
            state.umap.edges = [];
            const pts = state.umap.points;
            for (let i = 0; i < pts.length; i++) {
                for (let j = i + 1; j < pts.length; j++) {
                    const d = Math.hypot(pts[i].x - pts[j].x, pts[i].y - pts[j].y);
                    if (d < 0.45) {
                        state.umap.edges.push({ i, j, weight: 1 - d / 0.45 });
                    }
                }
            }


            // Update Python Snippet
            const pyCode = `import umap\nimport numpy as np\n\nmapper = umap.UMAP(\n    n_neighbors=${state.umap.nNeighbors},\n    min_dist=0.1,\n    metric="${state.umap.inputMetric}",\n    output_metric="${state.umap.outputMetric}",\n    random_state=42\n)\nembedding = mapper.fit_transform(X_high_dim)`;
            const pySnippet = document.getElementById('pythonCodeSnippet');
            if (pySnippet) pySnippet.textContent = pyCode;


            renderUmap();
        }


        function renderUmap() {
            if (!umapCtx || !umapCanvas) return;
            const w = umapCanvas.width;
            const h = umapCanvas.height;
            const center = { x: w / 2, y: h / 2 };
            const radius = Math.min(w, h) * 0.42;
            const isDark = document.documentElement.classList.contains('dark');


            umapCtx.clearRect(0, 0, w, h);


            if (state.umap.outputMetric === 'hyperboloid') {
                // Draw Poincaré Boundary Circle
                umapCtx.beginPath();
                umapCtx.arc(center.x, center.y, radius, 0, Math.PI * 2);
                umapCtx.strokeStyle = isDark ? '#ec4899' : '#db2777';
                umapCtx.lineWidth = 2 * window.devicePixelRatio;
                umapCtx.fillStyle = isDark ? 'rgba(236, 72, 153, 0.03)' : 'rgba(219, 39, 119, 0.03)';
                umapCtx.fill();
                umapCtx.stroke();
            }


            // Draw Fuzzy Edges
            umapCtx.strokeStyle = isDark ? 'rgba(59, 130, 246, 0.25)' : 'rgba(37, 99, 235, 0.25)';
            state.umap.edges.forEach(e => {
                const p1 = state.umap.points[e.i];
                const p2 = state.umap.points[e.j];
                umapCtx.lineWidth = e.weight * 2 * window.devicePixelRatio;
                umapCtx.beginPath();
                umapCtx.moveTo(center.x + p1.x * radius, center.y + p1.y * radius);
                umapCtx.lineTo(center.x + p2.x * radius, center.y + p2.y * radius);
                umapCtx.stroke();
            });


            // Draw Nodes
            const colors = ['#3b82f6', '#10b981', '#f59e0b', '#ec4899'];
            state.umap.points.forEach(p => {
                umapCtx.beginPath();
                umapCtx.arc(center.x + p.x * radius, center.y + p.y * radius, 5 * window.devicePixelRatio, 0, Math.PI * 2);
                umapCtx.fillStyle = colors[(p.cluster - 1) % colors.length];
                umapCtx.fill();
                umapCtx.strokeStyle = '#ffffff';
                umapCtx.lineWidth = 1.5 * window.devicePixelRatio;
                umapCtx.stroke();
            });
        }
